export function Color({ mode, tokens }) {

    return (
        <div>
            <p>These color tokens are prefixed by <code>primitive.color</code></p>
            <ColorPalette>
                {
                    tokens && Object.entries(tokens.primitive.color).map(([id, data]) => {
                        let colors;

                        if (data.$value) {
                            colors = { "": data.$value };
                        } else {
                            colors = Object.fromEntries(
                                Object.entries(data).map(([shade, obj]) => [shade, obj.$value])
                            );
                        }

                        return (
                            <ColorItem
                                key={id}
                                title={capitalize(id)}
                                subtitle={'primitive.color.' + id}
                                colors={colors}
                            />
                        );
                    })
                }
            </ColorPalette>
        </div>
    );
}

export function getAllTokensValueByNameFor(tokens, customType = undefined) {
    const flattenJsonGroup = []

    if (typeof tokens !== 'object' || tokens === null) {
        return flattenJsonGroup;
    }

    Object.entries(tokens).forEach(([key, value]) => {
        if(value.$value) {
            flattenJsonGroup.push({
                name: value.name,
                value: value.$value,
                originalValue: value?.original?.$value.replace(/[{}]/g, ''),
                type: customType ?? value.$type,
                description: value.$description
            });
        } else {
            flattenJsonGroup.push(...getAllTokensValueByNameFor(value, customType));
        }
    });

    return flattenJsonGroup;
}

export function TokensPreview(tokens, withDescription = true) {
    return <table style={{borderCollapse: "collapse", borderSpacing: 0, width: '100%', textAlign: 'left'}}>
        <thead>
            <tr>
                <th>Name</th>
                <th>Value</th>
                { withDescription && <th>Description</th> }
            </tr>
        </thead>
        <tbody>
            {
                tokens.map(token => (
                    <tr>
                        <td>{token.name}</td>
                        <td>
                            {TokenValue(token.value, token.originalValue, token.type)}
                        </td>
                        { withDescription && <td>{token.description}</td>}
                    </tr>
                ))
            }
        </tbody>
    </table>
}

export function TokenValue(value, originalValue, type) {
    if(type === 'color') {
        return (
            <div style={{
                display: 'flex',
                flexDirection: 'column',
                alignItems: 'center',
                gap: '0.5rem'
            }}>
                <div style={{width: '100%', height: '24px', backgroundColor: value, border: 'solid 1px black'}}></div>
                <code>{ originalValue }</code>
            </div>
        );
    } else if(type === 'spacing') {
        return (
            <div style={{display: "flex", flexDirection: 'row', gap: value}}>
                <div style={{width: '24px', height: '24px', backgroundColor: "var(--wcs-semantic-color-background-action-primary, #6558b1)", border: 'solid 1px black'}}></div>
                <div style={{width: '24px', height: '24px', backgroundColor: "var(--wcs-semantic-color-background-action-primary, rgba(0, 0, 0, 0))", border: 'solid 1px black'}}></div>
            </div>
        );
    } else if(type === 'motion') {
        return (
            MoveCircleInfinite(value)
        )
    }
    else {
        return value
    }
}

export function MoveCircleInfinite(duration) {
    const circleStyle = {
        width: '24px',
        height: '24px',
        backgroundColor: 'var(--wcs-semantic-color-background-action-primary, #6558b1)',
        border: 'solid 1px black',
        position: 'relative',
        animation: `moveCircleVertical ${duration} infinite`,
    };

    return (
        <>
            <style>
                {`
          @keyframes moveCircleVertical {
            0% {
                transform: rotate(0deg);
                transform-origin: center;
            }
            100% {
                transform: rotate(360deg);
                transform-origin: center;
            }
          }
        `}
            </style>
            <div style={circleStyle}></div>
        </>
    );
}

export default function DesignTokens() {
    const [mode, setMode] = useState('sncf-holding');

    let tokens;
    switch (mode) {
        case 'sncf-holding':
            tokens = tokensSncfHolding;
            break;
        case 'sncf-reseau':
            tokens = tokensSncfReseau;
            break;
        case 'sncf-voyageurs':
            tokens = tokensSncfVoyageurs;
            break;
        default:
            tokens = tokensSncfHolding;
    }

    const handleChange = (event) => {
        setMode(event.target.value);
    };

    return (
        <div>
            <div className="radio-group" style={{ marginBottom: '20px' }}>
                <label style={{ marginRight: '15px' }}>
                    <input
                        type="radio"
                        value="sncf-holding"
                        checked={mode === 'sncf-holding'}
                        onChange={handleChange}
                    />
                    SNCF Holding
                </label>
                <label style={{ marginRight: '15px' }}>
                    <input
                        type="radio"
                        value="sncf-reseau"
                        checked={mode === 'sncf-reseau'}
                        onChange={handleChange}
                    />
                    SNCF Réseau
                </label>
                <label>
                    <input
                        type="radio"
                        value="sncf-voyageurs"
                        checked={mode === 'sncf-voyageurs'}
                        onChange={handleChange}
                    />
                    SNCF Voyageurs
                </label>
            </div>
        </div>
    );
}
